func ArbolBin<Element>::levelOrderTraversal(): Lista<Lista<Element>>
var 
    Lista<Lista<Element>>: result
    Lista<Element>: nivel 
    Cola<PointerToNodo<ArbolBin<Element>>>: aux
    PointerToNodo<Element>: hijoIzq,hijoDer
    int: len
begin
    result.construir()
    aux.construir()
    if(instance.esNulo())then
        return result
    endif
    aux.encolar(instance.raiz)
    while(!aux.esVacia())do
        nivel.construir()
        len <- aux.getLength()
        for i <- 1 to len do
            nivel.insertar(aux.getHead()^.getInfo(),nivel.getLength()+1)
            hijoIzq <- aux.getHead()^.getHijoIzq()
            hijoDer <- aux.getHead()^.getHijoDer()
            if(hijoIzq)then
                aux.encolar(hijoIzq)
            endif
            if(hijoDer)then
                aux.encolar(hijoDer)
            endif
            aux.desencolar()
        endfor
        result.insertar(nivel,result.getLength()+1)
    endwhile
    return result
endfunc

func ArbolBin<string>::evaluateExpression(PointerToNodoBin<string>: ptr): int

    string: op
    int: a,b

    if(!ptr^.getHijoizq() && !ptr^.getHijoDer())then
        return stringToInt(ptr^.getInfo())
    else
        a <- instance.evaluateExpression(ptr^.getHijoizq())

        return operar(a,b,op)
    endif

func operar(int: a,b,string: op): int
var
    int: result
begin
    op <- stringToChar(op)
    switch(op)
        case '+'
            return a + b
            break
        case '-'
            return a - b
            break 
        case '*'
            return a * b 
            break 
        case '/'
            return a / b 
            break 
        default
            return 0
    endswitch
endfunc

func getDiameter(PointerToNodoN<int>: ptr,ref: int: maxDiam): int
var
    int: altura,max1,max2
    PointerToNodoN<int>: hijo
begin
    max1 <- 0
    max2 <- 0
    hijo <- ptr^.getHijoIzq()
    while(hijo)do
        altura <- 1 + instance.getDiameter(hijo,maxDiam)
        if(altura > max1)then
            max2 <- max1
            max1 <- altura
        else
            if(altura > max2)then
                max2 <- altura
            endif
        endif
        hijo <- hijo^.getHermanoDer()
    endwhile
    if(max1 + max2 > maxDiam)then
            maxDiam <- max1 + max2
        endif
    return max1
endfunc
