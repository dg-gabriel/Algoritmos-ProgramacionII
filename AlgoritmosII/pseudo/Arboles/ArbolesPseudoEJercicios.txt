proc ArbolBin<Element>::destruirNodos(ref: PointerToNodoBin<Element> ptrNodo)
begin
    if(!ptrNodo)then
        return
    endif
    if(!ptrNodo^.getHijoIzq() && !ptrNodo().getHijoDer)then
        free(ptrNodo)
        ptrNodo <- NULL
    else
        destruirNodos(ptrNodo^.getHijoIzq())
        destruirNodos(ptrNodo^.getHijoDer())
        free(ptrNodo)
        ptrNodo <- NULL
    endif
endproc

func ArbolBin<Element>::levelOrderTraversal(): Lista<Lista<Element>
var
    Cola<PointerToNodoBin<Element>>: aux
    Lista<Lista<Element>>: result
    Lista<Element>: sublist
    int: len,i
begin
    result.construir()
    if(instance.esNulo())then
        return result
    endif
    aux.encolar(instance.raiz)
    while(!aux.esVacia())do
        sublist.construir()
        len <- aux.getLength()
        for i <- 1 to len do
            if(aux.getHead()^.getHijoIzq())then
                aux.encolar(aux.getHead()^.getHijoIzq())
            endif
            if(aux.getHead()^.getHijoDer())then
                aux.encolar(aux.getHead^.getHijoDer())
            endif
            sublist.insertar(aux.getHead()^.getInfo())
            aux.desencolar()
        endfor
        result.insertar(sublist,result.getLength()+1)
    endwhile
    return result
endfunc
