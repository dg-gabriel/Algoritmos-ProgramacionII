proc GrafoDirigido<Element>::reverseGraph()
var
    Array[1..N] of PointerToNodoArco<Element>: listaViejos
    PointerToNodoArco<Element>: arcoAct,arcoSig
    PointerToNodoVertice<Element>: act,destino
    int: i
begin
    act <- instance.primero
    for i <- 1 to instance.getNvertices() do
        listaViejos[i] <- act^.getListaAdy()
        act^.setListaAdy(NULL)
        act <- act^.getSiguiente()
    endfor
    act <- instance.primero
    i <- 1
    while(act)do
        arcoAct <- listaViejos[i]
        while(arcoAct)do
            arcoSig <- arcoAct^.getSiguiente()
            destino <- arcoAct^.getDestino()
            arcoAct^.setDestino(act)
            arcoAct^.setSiguiente(destino^.getListaAdy())
            destino^.setListaAdy(arcoAct)
            arcoAct <- arcoSig
        endwhile
        i <- i + 1
        act <- act^.getSiguiente()
    endwhile
endproc

func BFS(Grafo<int>: g,int: origen,int: destino): Lista<int>
var
    Lista<int>: result,vecinos
    Array[1..N] of bool: visitados
    Array[1..N] of int: padres
    Cola<int>: colaAux
    int: v,w,padre
begin
    result.construir()
    for i <- 1 to g.getNvertices() do 
        visitados[i] <- false
    endfor
    visitados[origen] <- true
    padres[origen] <- 0
    colaAux.encolar(origen)
    while(!colaAux.esVacia() AND colaAux.getHead() != destino)do
        v <- colaAux.getHead()
        vecinos <- g.getVecinos(v)

                visitados[w] <- true
                padres[w] <- v
                colaAux.encolar(w)
            endif
            vecinos.eliminar(1)
        endwhile
        colaAux.desencolar()    
    endwhile
    if(!visitados[destino])then
        return result
    endif
    v <- destino
    while(v != 0)do
        result.insertar(v,1)
        v <- padres[v]
    endwhile
    return result
endfunc
