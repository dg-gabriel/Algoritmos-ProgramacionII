Desarrolle el método de la clase GrafoNoDirigido:
func GrafoNoDirigido<Elemento>::countIsolated(): int
Que retorne el número de nodos aislados (que no están conectados con ningún otro). Su solución debe
ser a lo sumo O(n + m) DEBE USAR APUNTADORES EN SU SOLUCIÓN

func GrafoNoDir<Element>::countIsolated(): int
var 
    PointerToNodoVertice<Element>: act
    int: cont 
begin
    cont <- 0
    act <- instance.primero
    while(act)do
        if(!act^.getListaAdy())then
            cont <- cont + 1
        endif
        act <- act^.getSiguiente()
    endwhile
    return cont
endfunc

Dado un grafo no dirigido, no ponderado y mapeado, cree la función::
func createsCycle(Grafo<int>: g, int: u, int: v): bool
Que determine si al agregar el arco (u, v) al grafo se formaría un ciclo. Valor 7 puntos

func createsCycle(Grafo<int>: g,int: u,int: v): bool
var
    Array[1..N] of bool: visitados
    int: i
    bool: tieneCiclo 
begin
    tieneCiclo <- false
    if(!g.existeVertice(u) || !g.existeVertice(v))then
        return tieneCiclo
    endif
    g.insertarArco(u,v,0)
    for i <- 1 to g.getNVertices() do
        visitados[i] <- false
    endfor
    for i <- to g.getNVertices() do
        if(!visitados[i])then
            createsCycle(g,i,0,visitados,tieneCiclo)
        endif
    endfor
    return tieneCiclo
endfunc

proc createsCycle(Grafo<int>: g,int: v,ref: int: padre,ref: Array[1..N] of bool: visitados,ref: bool: found)
var
    int: w 
    Lista<int>: vecinos
begin
    visitados[v] <- true
    vecinos <- g.getVecinos(v)
    while(!vecinos.esVacia())do
        w <- vecinos.consultar(1)
        if(!visitados[w])then
            createsCycle(g,w,v,visitados)
        else
            if(w != padre)then
                found <- true
            endif
        endif
        vecinos.eliminar(1)
    endwhile
endproc

// tiene exactamente el problema que suponia, busca en todo el grafo en lugar de buscar si u y v completan un ciclo,
// puede dar falso positivo si hay otros ciclos en el grafo
// lo correcto es hacer un dfs empezando desde u y si se encuentra a v, retornar true

func createsCycle(Grafo<int>: g,int: u,v): bool
var
    bool: creaCiclo
    Array[1..N] of bool: visitados
    int: i
begin
    if(!g.existeVertice(u) || g.existeVertice(v))then
        return false
    endif
    for i <- 1 to g.getNvertices() do
        visitados[i] <- false
    endfor
    creaCiclo <- createsCycle(g,u,v,visitados)
    return creaCiclo
endfunc

func createsCycle(Grafo<int>: g,int: origen,int: destino,ref: Array[1..N] of bool: visitados): bool
var
    Lista<int>: vecinos
    Pila<int>: dfs
    int: v,w
begin
    dfs.apilar(origen)
    while(!dfs.esVacia())do
        v <- dfs.desapilar()
        if(!visitados[v])then
            visitados[v] <- true
            vecinos <- g.getVecinos(v)
            while(!vecinos.esVacia())do
                w <- vecinos.consultar(1)
                if(w == destino)then
                    return true
                endif
                if(!visitados[w])then
                    dfs.apilar(w)
                endif
                vecinos.eliminar(1)
            endwhile
        endif
    endwhile
    return false
endfunc
