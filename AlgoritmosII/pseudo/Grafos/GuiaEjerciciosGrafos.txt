incidentes).

Implemente una estructura Grafo que permita agregar nodos dinámicamente y conectar nodos mediante aristas.
proc Grafo::agregarNodo(int: v)
var
    PointerToNodoVertice<int>: nuevo
begin
    nuevo <- new(NodoVertice<int>)
    nuevo.construir(v,instance.primero,NULL)
    instance.primero <- nuevo
    nVertices <- nVertices + 1
endproc

proc Grafo::agregarArista(int: a,int: b)
var
    PointerToNodoArco<int>: nuevoArco
    PointerToNodVertice<int>: ptrV,ptrW,ptr
begin
    ptr <- instance.primero
    while(ptr)do 
        if(ptr^.getInfo() = a)then
            ptrV <- ptr
        else if(ptr^.getInfo() = b)then
                ptrW <- ptr 
            endif
        endif
    endwhile
    if(!ptrW || !ptrV)then
        return
    endif
    nuevoArco <- new(NodoArco<int>)
    ptrV^.setListaAdy(nuevoArco.construir(ptrW,NULL))
    return
endfunc


Dado un grafo no dirigido, determine cuántas componentes conexas existen. Puede apoyarse en BFS o DFS.
func componentesConexas(Grafo<int>: g): int
var
    int: u,v,w,i,nroComp
    Lista<int>: vertices,vecinos
    Array[1..N] of bool: visitados
    Cola<int>: cola 
begin
    nroComp <- 0
    for i <- 1 to g.getNVertices() do
        visitados[i] <- false
    endfor
    vertices.construir()
    vertices <- g.getVertices()
    while(!vertices.esVacia())do
        u <- vertices.consultar(1)
        if(visitados[u])then
            vertices.eliminar(1)
        else
            nroComp <- nroComp + 1
            visitados[u] <- true
            cola.construir()
            cola.encolar(u)
            while(!cola.esVacia())
                v <- cola.getHead()
                vecinos <- g.getVecinos(v)
                while(!vecinos.esVacia())do
                    w <- vecinos.consultar(1)
                    if(!visitados[w])then
                        visitados[w] <- true
                        cola.encolar(w)
                    endif
                    vecinos.eliminar(1)
                endwhile
                cola.desencolar()
            endwhile
        endif
    endwhile
    return nroComp
endfunc


Modele una red social como un grafo no dirigido donde cada nodo representa una persona y cada arista representa una
amistad.

proc redAmistad::agregarPersona(string: nombre)
var
    PointerToNodoVertice<string>: nuevaPersona
begin
    nuevaPersona <- new(NodoVertice<string>)
    nuevaPersona.construir(nombre,instance.primero,NULL)
    instance.primero <- nuevaPersona
    nVertices <- nVertices + 1
endproc

proc redAmistad::agregarAmistad(string: nombre1,nombre2)
var
    PointerToNodoArco<string>: arcoU,arcoV
    PointerToNodoVertice<string>: ptrU,ptrV,ptr
begin
    ptrU <- NULL
    ptrV <- NULL
    ptr <- instance.primero
    while(ptr)do
        if(ptr^.getInfo() = nombre1)
            ptrU <- ptr 
        else if(ptr^.getInfo() = nombre2)
                ptrV <- ptr
            endif
        endif
        ptr <- ptr^.getSiguiente()
    endwhile
    if(!ptrU || !ptrV)then
        return
    endif
    arcoU <- new(NodoArco<string>)
    arcoU^.construir(ptrV,ptrU.^getListaAdy(),0)
    ptrU^.setListaAdy(arcoU)

    arcoV <- new(NodoArco<string>)
    arcoV^.construir(ptrU,ptrV^.getListaAdy(),0)
    ptrV^.setListaAdy(arcoV)

endproc

func nivelDeConexion(string: persona1,persona2): int
var
    int: nivelAct
    Cola<PointerToNodoVertice<string>>: colaNodos
    Cola<int>: colaNiveles
    Lista<string>: visitados
    PointerToNodoVertice<string>: actual,destino,ptrA
    PointerToNodoArco<string>: arco
begin
    ptrA <- NULL
    actual <- instance.first
    while(ptr)do
        if(ptr^.getInfo() = persona1)then
            ptrA <- ptr 
        endif
        ptr <- ptr^.getSiguiente()
    endwhile
    if(!ptrA)then
        return nivel 
    endif
    colaNodos.construir()
    colaNiveles.construir()
    visitados.construir()
    colaNodos.encolar(ptrA)
    colaNiveles.encolar(0)
    visitados.insertar(ptrA^.getInfo(),1)
    while(!colaNodos.esVacia())do
        actual <- colaNodos.getHead()
        nivelAct <- colaNiveles.getHead()
        if(actual^.getInfo() = persona2)then
            return nivelAct
        endif
        arco <- actual.getListaAdy()
        while(arco)do
            destino <- arco^.getInfo()
            if(!visitados.buscar(destino^.getInfo()))then
                visitados.insertar(destino^.getInfo(),1)
                colaNodos.encolar(destino)
                colaNiveles.encolar(nivelAct+1)
            endif
            arco <- arco^.getSiguiente()
            // asi se recorren vertices adyacentes con punteros
        endwhile
        colaNodos.desencolar()
        colaNiveles.desencolar()
    endwhile
    return -1
endfunc


Modele el sistema de materias de una universidad como un grafo dirigido, donde una arista A → B indica que A es
prerrequisito de B

proc SistemaMaterias::agregarMateria(string: materia)
var
    PointerToNodoVertice<string>: nuevaMat
begin
    nuevaMat <- new(NodoVertice<string>)
    nuevaMat^.construir(nombre,instance.primero,NULL)
    instance.primero <- nuevaMat
    nVertices <- nVertices + 1
endproc

proc SistemaMaterias::agregarPrerequisito(string: a,b)
var
    PointerToNodoVertice<string>: ptrA,ptrB,act
    PointerToNodoArco<string>: prelacion
begin
    ptrA <- NULL
    ptrB <- NULL
    act <- instance.primero
    while(act)do
        if(act^.getInfo() = a)then
            ptrA <- act
        else if(act^.getInfo() = b)then 
                ptrB <- act
            endif
        endif
        act <- act^.getSiguiente()
    endwhile
    if(!ptrA || !ptrB)then
        return
    endif
    prelacion <- new(NodoArco<string>)
    prelacion^.construir(ptrB,ptrA^.getListaAdy(),0)
    ptrA.setListaAdy(prelacion)
endproc

func puedeInscribirse(string: materia,Lista<string>: aprobadas): bool
var
    PointerToNodoArco<string>: arcoAct
    PointerToNodoVertice: act
    bool: requisito
begin
    act <- instance.first
    while(act)do
        arcoAct <- act^.getListaAdy()
        while(arcoAct)do
            if(arcoAct^.getInfo() = materia)
                requisito <- false
                if(aprobadas.contiene(act^.getInfo()))then
                    // aprobo la prelacion inmediata
                    requisito <- true
                endif
                if(!requisito)then
                    return false
                endif
            endif
            arcoAct <- arcoAct^.getSiguiente()
        endwhile
        act <- act^.getSiguiente()
    endwhile
    return true
endfunc

Dado un grafo dirigido, determine si existe algún ciclo en el grafo utilizando DFS. Considere el estado de cada nodo
durante el recorrido

func tieneCiclo(Grafo<Element>: g): bool
var
    Array[1..N] of bool: visitados
    Lista<Element>: vecinos
    Pila<Element>: pila
    int: i, n
    Element: v
begin
    result.construir()
    pila.construir()
    n <- g.getNVertices()

    for i <- 1 to n do
        visitados[i] <- false
    endfor
    
    pila.apilar(g.getPrimero())
    while(¬pila.esVacia())do
        v <- pila.getTop()
        pila.desapilar()
        
        if(¬visitados[v])then
            visitados[v] <- true
            result.insertar(v, result.getLength() + 1)
            
            vecinos <- g.getVecinos(v)
            
            while(¬vecinos.esVacia())do
                pila.apilar(vecinos.consultar(1))
                vecinos.eliminar(1)
            endwhile
        else
            // si el nodo ya fue visitado en un grafo dirigido, significa que hay ciclo
            return true
    endwhile
    
    return false
endfunc
