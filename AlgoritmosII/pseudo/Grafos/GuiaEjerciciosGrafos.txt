Dado un grafo no dirigido, determine cuántas componentes conexas existen. Puede apoyarse en BFS o DFS.
func componentesConexas(Grafo<int>: g): int
var
    int: u,v,w,i,nroComp
    Lista<int>: vertices,vecinos
    Array[1..N] of bool: visitados
    Cola<int>: cola 
begin
    nroComp <- 0
    for i <- 1 to g.getNVertices() do
        visitados[i] <- false
    endfor
    vertices.construir()
    vertices <- g.getVertices()
    while(!vertices.esVacia())do
        u <- vertices.consultar(1)
        if(visitados[u])then
            vertices.eliminar(1)
        else
            nroComp <- nroComp + 1
            visitados[u] <- true
            cola.construir()
            cola.encolar(u)
            while(!cola.esVacia())
                v <- cola.getHead()
                vecinos <- g.getVecinos(v)
                while(!vecinos.esVacia())do
                    w <- vecinos.consultar(1)
                    if(!visitados[w])then
                        visitados[w] <- true
                        cola.encolar(w)
                    endif
                    vecinos.eliminar(1)
                endwhile
                cola.desencolar()
            endwhile
        endif
    endwhile
    return nroComp
endfunc

Dado un grafo dirigido, determine si existe algún ciclo en el grafo utilizando DFS. Considere el estado de cada nodo

func tieneCiclo(Grafo<Element>: g): bool
var
    Array[1..N] of bool: visitados
    Lista<Element>: vecinos
    Pila<Element>: pila
    int: i, n
    Element: v
begin
    result.construir()
    pila.construir()
    n <- g.getNVertices()

    for i <- 1 to n do
        visitados[i] <- false
    endfor
    
    pila.apilar(g.getPrimero())
    while(¬pila.esVacia())do
        v <- pila.getTop()
        pila.desapilar()
        
        if(¬visitados[v])then
            visitados[v] <- true
            result.insertar(v, result.getLength() + 1)
            
            vecinos <- g.getVecinos(v)
            
            while(¬vecinos.esVacia())do
                pila.apilar(vecinos.consultar(1))
                vecinos.eliminar(1)
            endwhile
        else
            // si el nodo ya fue visitado en un grafo dirigido, significa que hay ciclo
            return true
    endwhile
    
    return false
endfunc

Dado un grafo dirigido acíclico (DAG), implemente el algoritmo de ordenamiento topológico. El resultado debe ser una
lista de nodos donde cada nodo aparece antes que sus dependientes

func topologicalSort(Grafo<int>: g): Lista<int>
var
    Lista<int>: result
    int: n
    Array[1..N] of bool: visitados
begin
    result.construir()
    n <- g.getVertices()
    if(g.esNulo())then
        return result
    endif
    for i <- 1 to n do
        visitados[i] <- false
    endfor
    for i <- 1 to n do
        if(!visitados[i])then
            topologicalSort(g,i,visitados,result)
        endif
    endfor
    return result
endfunc

proc topologicalSort(Grafo<int>: g,int act,ref: Array[1..n] of bool: visitados,ref: Lista<int>: result)
var
    Lista<int>: sucesores
    int: v
begin
    visitados[act] <- true
    sucesores <- g.getSucesores(act)
    while(!sucesores.esVacia())do
        v <- sucesores.consultar(1)
        if(!visitados[v])then
            visitados[v] <- true
            topologicalSort(g,v,visitados,result)
        endif
        sucesores.eliminar(1)
    endwhile
    result.insertar(act,1)
endproc

Implemente el algoritmo de Dijkstra para encontrar las distancias mínimas desde un nodo origen hacia todos los demás
nodos del grafo dirigido con pesos positivos. Debe utilizar una cola de prioridad.

// se asume la estructura ColaPrioridad<Element> con el metodo insertar(elem,prioridad)
func dijkstra(Grafo<int>: g,int: origen): Array[1..n] of float
var
    Array[1..n] of float: distancias
    Array[1..n] of bool: visitados
    ColaPrioridad<int>: cola
    Lista<int>: vecinos
    float: costoAct
    int: i,v,w
begin
    for i <- 1 to g.getNvertices() do
        distancias[i] <- 999
        visitados[i] <- false
    endfor
    distancias[origen] <- 0
    cola.insertar(origen,0)
    while(!cola.esVacia())do
        v <- cola.getHead()
        cola.desencolar()
        if(!visitados[v])then
            visitados[v] <- true
            vecinos <- g.getSucesores(v)
            while(!vecinos.esVacia())do
                w <- vecinos.consultar(1)
                if(!visitados[w])then
                    costoAct <- distancias[v] + g.getCostoArco(v,w)
                    if(costoAct < distancias[w])then
                        distancias[w] <- costoAct
                        cola.insertar(w,distancias[w])
                    endif
                endif
                vecinos.eliminar(1)
            endwhile
        endif
    endwhile
    return distancias
endfunc

func puentes(Grafo<int>: g): Lista<Lista<int>>
var
    Lista<Lista<int>: result
    Array[1..N] of bool: visitados
    Array[1..N] of int: disc 
    Array[1..N] of int: low
    int: i,pasos
begin
    result.construir()
    for i <- 1 to g.getNVertices() do 
        visitados[i] <- false 
    endfor
    pasos <- 1
    for i <- 1 to g.getNvertices() do
        if(!visitados[i])then
            puentesDFS(g,i,0,visitados,disc,low,pasos,result)
        endif
    endfor
    return result
endfunc

proc puentesDFS(Grafo<int>: g,int: v,int: padre,ref: Array[1..N] of bool: visitados,ref: Array[1..N] of int: disc,ref: Array[1..N] of int: low,ref; int: pasos,ref:Lista<Lista<int>>: result)
var 
    Lista<int>: vecinos,arco
    int: w
begin
    vecinos.construir()
    visitados[v] <- true
    pasos <- pasos + 1
    disc[v] <- pasos 
    low[v] <- pasos
    vecinos <- g.getVecinos()
    while(!vecinos.esVacia())do
        w <- vecinos.consultar(1)
        if(!visitados[w])then
            puentesDFS(g,w,v,visitados,disc,low,pasos,result)
            low[v] <- min(low[v],low[w])
            if(low[w] > disc[v])then
                arco.construir()
                arco.insertar(v,1)
                arco.insertar(w,2)
                result.insertar(arco,result.getLength()+1)
            endif
        else
            if(w != padre)then
                low[v] <- min(low[v],disc[w])
            endif
        endif
        vecinos.eliminar(1)
    endwhile
endproc

func simularFallo(Grafo<string>: grafo,string: sub)
var
    Grafo<int>: g 
    Arrray[1..N] of string: nombres
    Array[1..N] of bool: visitados
    int: i,inicial,u,v,w,nroComp
    Lista<int>: vertices,vecinos
    Cola<int>: cola
begin
    g <- mapearGrafo(grafo,nombres)
    for i <- 1 to g.getNVertices() do
        visitados[i] <- false
    endfor
    inicial <- 0
    for i <- 1 to nombres.length() do
        if(nombres[i] = sub)then
            inicial = i
        endif
    endfor
    if(!inicial)then
        return 0
    endif
    visitados[inicial] <- true
    nroComp <- 0
    vertices <- g.getVertices()
    while(!vertices.esVacia())do
        u <- vertices.consultar(1)
        if(visitados[u])then
            vertices.eliminar(1)
        else
            nroComp <- nroComp + 1
            visitados[u] <- true
            cola.construir()
            cola.encolar(u)
            while(!cola.esVacia())do
                v <- cola.getHead()
                vecinos <- g.getVecinos(v)
                while(!vecinos.esVacia())do
                    w <- vecinos.consultar(1)
                    if(!visitados[w])then
                        visitados[w] <- true
                        cola.encolar(w)
                    endif
                    vecinos.eliminar(1)
                endwhile
                cola.desencolar()
            endwhile
        endif
    endwhile
    return nroComp
endfunc


Dado un grafo dirigido, implemente el Algoritmo de Tarjan para encontrar todas las componentes fuertemente conexas
(SCC). Una SCC es un subconjunto donde cada nodo es alcanzable desde cualquier otro dentro del mismo subconjunto

// asume grafo mapeado
func tarjan(Grafo<int>: g): Lista<Lista<int>>
var
    Array[1..N] of bool: visitados,enPila
    Array[1..N] of int: disc,low
    Pila<int>: enCurso
    Lista<Lista<int>>: result
    int: i,pasos
begin
    result.construir()
    for i <- 1 to g.getNVertices() do
        visitados[i] <- false
        enPila[i] <- false
    endfor
    pasos <- 0
    for i <- 1 to g.getNVertices() do 
        if(!visitados[i])then
            tarjan(g,i,enCurso,visitados,enPila,disc,low,result)
        endif
    endfor
    return result
endfunc

proc tarjan(Grafo<int>: g,int: v,ref: int: pasos,ref: Pila<int>: enCurso,ref: Array[1..N] of bool: visitados,ref: Array[1..N] of bool: enPila, ref: Array[1..N] of int: disc,ref: Array[1..N] of int: low,ref: Lista<Lista<int>: result)
var
    int: w,x
    Lista<int>: vecinos,sublist
begin
    visitados[v] <- true
    enCurso.apilar(v)
    enPila[v] <- true
    pasos <- pasos + 1
    disc[v] <- pasos
    low[v] <- pasos
    vecinos <- g.getSucesores(v)
    while(!vecinos.esVacia())do
        w <- vecinos.consultar(1)
        if(!visitados[w])then
            tarjan(g,w,pasos,enCurso,visitados,enPila,disc,low,result)
            low[v] <- min(low[w],low[v])
        endif
        if(enPila[w])then
            low[v] <- min(low[v], disc[w])
        endif
        vecinos.eliminar(1)
    endwhile
    if(low[v] = disc[v])then
        sublist.construir()
        x <- -1
        while(x != v)do
            x <- enCurso.getHead()
            enPila[x] <- false
            enCurso.desapilar()
            sublist.insertar(x,sublist.getLength()+1)
        endwhile
        result.insertar(sublist,result.getLength()+1)
    endif
endproc

// EJ. extras

func Grafo<Element>::mapearGrafo(ref: Array[1..N] of Element: posiciones): Grafo<int>
var 
    PointerToNodoVertice<Element>: actViejo
    PointerToNodoVertice<int>: nuevoVer,actNuevo
    PointerToNodoArco<Element>: arcoAct
    PointerToNodoArco<int>: nuevoArc
    Grafo<int>: nuevoGrafo
    int: i
begin
    nuevoGrafo.construir()
    actViejo <- instance.primero
    for i <- 1 to nVertices do
        posiciones[i] <- actViejo^.getInfo()
        actViejo <- actViejo^.getSiguiente()
    endfor
    for i <- nVertices in -1 to 1 do 
        nuevoVer <- new(NodoVertice<int>)
        nuevoVer^.construir(i,nuevoGrafo.primero,NULL)
        nuevoGrafo.primero <- nuevoVer
    endfor
    actViejo <- instance.primero
    actNuevo <- nuevoGrafo.primero
    while(actViejo && actNuevo)do
        arcoAct <- actViejo^.getListaAdy()
        while(arcoAct)do
            i <- buscarEnVector(posiciones,arcoAct^.getDestino()^.getInfo())
            nuevoArc <- new(NodoArco<int>)
            nuevoVer <- buscarVertice(nuevoGrafo,i)
            nuevoArc^.construir(nuevoVer,actNuevo^.getListaAdy(),arcoAct^.getPeso())
            actNuevo^.setListaAdy(nuevoArc)
            arcoAct <- arcoAct^.getSiguiente()
        endwhile
        actViejo <- actViejo^.getSiguiente()
        actNuevo <- actNuevo^.getSiguiente()
    endwhile
    nuevoGrafo.nVertices <- instance.nVertices
    nuevoGrafo.nArcos <- instance.nArcos
    return nuevoGrafo
endfunc

// N como longitud del arreglo
func buscarEnVector(Array[1..N] of Element: arreglo,Element: v): int 
var
    int: i
begin
    for i <- 1 to N do
        if(arreglo[i] = v)then
            return i 
        endif
    endfor
    return -1
endfunc

func buscarVertice(Grafo<int>: g,int: v): PointerToNodoVertice<int>
var 
    PointerToNodoVertice<int>: act 
begin
    act <- g.primero
    while(act)do
        if(act^.getInfo() = v)then
            return act
        endif
        act <- act^.getSiguiente()
    endwhile
    return NULL
endfunc
