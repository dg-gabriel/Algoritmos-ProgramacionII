Implemente una estructura Grafo que permita agregar nodos dinámicamente y conectar nodos mediante aristas.
proc Grafo::agregarNodo(int: v)
var
    PointerToNodoVertice<int>: nuevo
begin
    nuevo <- new(NodoVertice<int>)
    nuevo.construir(v,instance.primero,NULL)
    instance.primero <- nuevo
    nVertices <- nVertices + 1
endproc

proc Grafo::agregarArista(int: a,int: b)
var
    PointerToNodoArco<int>: nuevoArco
    PointerToNodVertice<int>: ptrV,ptrW,ptr
begin
    ptr <- instance.primero
    while(ptr)do 
        if(ptr^.getInfo() = a)then
            ptrV <- ptr
        else if(ptr^.getInfo() = b)then
                ptrW <- ptr 
            endif
        endif
    endwhile
    if(!ptrW || !ptrV)then
        return
    endif
    nuevoArco <- new(NodoArco<int>)
    ptrV^.setListaAdy(nuevoArco.construir(ptrW,NULL))
    return
endfunc


Dado un grafo no dirigido, determine cuántas componentes conexas existen. Puede apoyarse en BFS o DFS.
func componentesConexas(Grafo<int>: g): int
var
    int: u,v,w,i,nroComp
    Lista<int>: vertices,vecinos
    Array[1..N] of bool: visitados
    Cola<int>: cola 
begin
    nroComp <- 0
    for i <- 1 to g.getNVertices() do
        visitados[i] <- false
    endfor
    vertices.construir()
    vertices <- g.getVertices()
    while(!vertices.esVacia())do
        u <- vertices.consultar(1)
        if(visitados[u])then
            vertices.eliminar(1)
        else
            nroComp <- nroComp + 1
            visitados[u] <- true
            cola.construir()
            cola.encolar(u)
            while(!cola.esVacia())
                v <- cola.getHead()
                vecinos <- g.getVecinos(v)
                while(!vecinos.esVacia())do
                    w <- vecinos.consultar(1)
                    if(!visitados[w])then
                        visitados[w] <- true
                        cola.encolar(w)
                    endif
                    vecinos.eliminar(1)
                endwhile
                cola.desencolar()
            endwhile
        endif
    endwhile
    return nroComp
endfunc

proc redAmistad::agregarPersona(string: nombre)

    nVertices <- nVertices + 1
endproc

proc redAmistad::agregarAmistad(string: nombre1,nombre2)
var
    PointerToNodoArco<string>: arcoU,arcoV
    PointerToNodoVertice<string>: ptrU,ptrV,ptr
begin
    ptrU <- NULL
    ptrV <- NULL
    ptr <- instance.primero
    while(ptr)do
        if(ptr^.getInfo() = nombre1)
            ptrU <- ptr 
        else if(ptr^.getInfo() = nombre2)
                ptrV <- ptr
            endif
        endif
        ptr <- ptr^.getSiguiente()
    endwhile
    if(!ptrU || !ptrV)then
        return
    endif
    arcoU <- new(NodoArco<string>)
    arcoU^.construir(ptrV,ptrU.^getListaAdy(),0)
    ptrU^.setListaAdy(arcoU)

    arcoV <- new(NodoArco<string>)
    arcoV^.construir(ptrU,ptrV^.getListaAdy(),0)
    ptrV^.setListaAdy(arcoV)

endproc

func nivelDeConexion(string: persona1,persona2): int
var
    int: nivelAct
    Cola<PointerToNodoVertice<string>>: colaNodos
    Cola<int>: colaNiveles
    Lista<string>: visitados
    PointerToNodoVertice<string>: actual,destino,ptrA
    PointerToNodoArco<string>: arco
begin
    ptrA <- NULL
    actual <- instance.first
    while(ptr)do
        if(ptr^.getInfo() = persona1)then
            ptrA <- ptr 
        endif
        ptr <- ptr^.getSiguiente()
    endwhile
    if(!ptrA)then
        return nivel 
    endif
    colaNodos.construir()
    colaNiveles.construir()
    visitados.construir()
    colaNodos.encolar(ptrA)
    colaNiveles.encolar(0)
    visitados.insertar(ptrA^.getInfo(),1)
    while(!colaNodos.esVacia())do
        actual <- colaNodos.getHead()
        nivelAct <- colaNiveles.getHead()
        if(actual^.getInfo() = persona2)then
            return nivelAct
        endif
        arco <- actual.getListaAdy()
        while(arco)do
            destino <- arco^.getInfo()
            if(!visitados.buscar(destino^.getInfo()))then
                visitados.insertar(destino^.getInfo(),1)
                colaNodos.encolar(destino)
                colaNiveles.encolar(nivelAct+1)
            endif
            arco <- arco^.getSiguiente()
            // asi se recorren vertices adyacentes con punteros
        endwhile
        colaNodos.desencolar()
        colaNiveles.desencolar()
    endwhile
    return -1
endfunc

    nuevaMat <- new(NodoVertice<string>)

    instance.primero <- nuevaMat
    nVertices <- nVertices + 1
endproc

proc SistemaMaterias::agregarPrerequisito(string: a,b)
var
    PointerToNodoVertice<string>: ptrA,ptrB,act
    PointerToNodoArco<string>: prelacion
begin
    ptrA <- NULL
    ptrB <- NULL
    act <- instance.primero
    while(act)do
        if(act^.getInfo() = a)then
            ptrA <- act
        else if(act^.getInfo() = b)then 
                ptrB <- act
            endif
        endif
        act <- act^.getSiguiente()
    endwhile
    if(!ptrA || !ptrB)then
        return
    endif
    prelacion <- new(NodoArco<string>)
    prelacion^.construir(ptrB,ptrA^.getListaAdy(),0)
    ptrA.setListaAdy(prelacion)
endproc

func SistemaMaterias::puedeInscribirse(string: materia, Lista<string>: aprobadas): bool
var
    PointerToNodoArco<string>: arcoAct
    PointerToNodoVertice<string>: act // Faltó el <string> en tu declaración
    bool: requisitoAprobado
    int: i
begin
    act <- instance.primero
    while (act) do
        arcoAct <- act^.getListaAdy()
        
        while (arcoAct) do
            if (arcoAct^.getInfo() = materia) then
                requisitoAprobado <- false
                for i <- 1 to aprobadas.getLength() do
                    if (aprobadas.consultarPos(i) = act^.getInfo()) then
                        requisitoAprobado <- true
                    endif
                endfor
                
                if (¬requisitoAprobado) then
                    return false
                endif
            endif
            arcoAct <- arcoAct^.getSiguiente()
        endwhile
        
        act <- act^.getSiguiente()
    endwhile
    return true
endfunc


Dado un grafo dirigido, determine si existe algún ciclo en el grafo utilizando DFS. Considere el estado de cada nodo
durante el recorrido.

// se asume grafo mapeado
func tieneCiclo(Grafo<Element>: g): bool
var
    Array[1..N] of int: estado
    int: i, n
begin
    n <- g.getVertices()
    for i <- 1 to n do
        estado[i] <- 0
    endfor
    for i <- 1 to n do
        if(!estado[i])then 
            if(tieneCiclo(g,i,estado))then
                return true
            endif
        endif
    endfor
    return false
endfunc

        v <- pila.getTop()
        if(estado[v] = 0)then
            estado[v] <- 1     
            vecinos <- g.getVecinos(v)     
            while(¬vecinos.esVacia())do
                w <- vecinos.consultar(1)
                if(estado[w] = 1)then
                    return true
                else if(estadp[w] = 0)then
                    pila.apilar(w)
                endif
                vecinos.eliminar(1)
            endwhile

            else
                pila.desapilar()

Modele un sistema de recomendación como un grafo no dirigido donde los nodos son productos y las aristas indican que
dos productos suelen comprarse juntos.

// se asume Producto como su propio tipo de dato

proc SistemaRecomendacion::agregarProducto(Producto: p)
var
    PointerToNodoVertice<Producto>: nuevoProd
begin
    nuevoProd <- new(NodoVertice<Producto>)
    nuevoProd^.construir(p,instance.primero,NULL)
    instance.primero <- nuevoProd
    nVertices <- nVertices + 1
endproc

proc SistemaRecomendacion::agregarRelacion(Producto: p1,p2)
var
    PointerToNodoVertice<Producto> ptr1,ptr2,act
    PointerToNodoArco<Producto>: arco1,arco2

begin
    ptr1 <- NULL
    ptr2 <- NULL
    act <- instance.primero
    while(act)do
        if(act^.getInfo() = p1)then
            ptr1 <- act
        else if(act^.getInfo() = p2)then
                ptr2 <- act 
            endif
        endif
        act <- act.getSiguiente()
    endwhile
    if(!ptr1 || !ptr2)then
        return
    endif

    arco1 <- new(NodoArco<Producto>)
    arco1^.construir(ptr1,ptr2^.getListaAdy(),0)
    ptr1^.setListaAdy(arco1)

    arco2 <- new(NodoArco<Producto>)
    arco2^.construir(ptr2,ptr1^.getListaAdy(),0)
    ptr2^.setListaAdy(arco2)
endproc

func SistemaRecomendacion::recomendar(Producto: p): Lista<Producto>
var
    Lista<Producto>: result
    PointerToNodoVertice<Producto>: prod,act
    PointerToNodoArco<Producto>: adyacentes
begin
    result.construir()
    prod <- NULL
    adyacentes <- NULL
    act <- instance.primero
    while(act)do
        if(act^.getInfo() = p)then
            prod <- act
        endif
        act <- act^.getSiguiente()
    endwhile
    if(prod)then
        adyacentes <- prod^.getListaAdy()
    endif
    while(adyacentes)do
        result.insertar(adyacentes^.getInfo(),result.getLength()+1)
        adyacentes <- adyacentes^.getSiguiente()
    endwhile
    return result
endfunc

// se asume un mapeo del grafo
func SistemaRecomendacion::recomendarExtendido(Producto: p): Lista<Producto>
var
    PointerToNodoVertice<Producto>: actual,ptrP
    PointerToNodoArco<Producto>: arcoAct
    Cola<PointerToNodoVertice<Producto>>: nivel1,nivel2
    Lista<Producto>: result
begin
    ptrP <- NULL
    result.construir()
    actual <- instance.primero
    while(actual)do
        if(actual^.getInfo() = p)then
            ptrP <- actual
            break
        endif
        actual <- actual^.getSiguiente()
    endwhile
    if(!actual)then
        return result
    endif
    nivel1.construir()
    arcoAct <- ptrP^.getListaAdy()
    while(arcoAct)do
        nivel1.encolar(arcoAct^.getInfo())
        result.insertar(arcoAct^.getInfo()^.getInfo(),result.getLength()+1)
        arcoAct <- arcoAct^.getSiguiente()
    endwhile
    while(!nivel1.esVacia())do
        arcoAct <- nivel1.getHead()^.getListaAdy()
        nivel1.desencolar()
        while(arcoAct)do
            if(!(result.contiene(arcoAct^.getInfo()^.getInfo())))then
                result.insertar(arcoAct^.getInfo()^.getInfo())
            endif
            arcoAct <- arcoAct^getSiguiente()
        endwhile
    endwhile
    return result
endfunc
